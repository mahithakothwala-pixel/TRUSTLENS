# ---------- DATABASE ----------

def initialize_database():
    with sqlite3.connect(DB_NAME) as connection:
        connection.execute("""
            CREATE TABLE IF NOT EXISTS checks (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                created_at TEXT,
                check_type TEXT,
                content TEXT,
                verdict TEXT,
                explanation TEXT
            )
        """)

def save_check(check_type, content, verdict, explanation):
    with sqlite3.connect(DB_NAME) as connection:
        connection.execute(
            """INSERT INTO checks
            (created_at, check_type, content, verdict, explanation)
            VALUES (?, ?, ?, ?, ?)""",
            (
                datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                check_type,
                content[:2000],
                verdict,
                explanation
            )
        )

def get_history():
    with sqlite3.connect(DB_NAME) as connection:
        return connection.execute(
            """SELECT created_at, check_type, content, verdict, explanation
            FROM checks ORDER BY id DESC LIMIT 100"""
        ).fetchall()

initialize_database()

# ---------- ANALYSIS FUNCTIONS ----------

WARNING_PATTERNS = {
    "Urgent pressure": [
        "act immediately",
        "forward immediately",
        "urgent",
        "within 10 minutes",
        "account will be blocked",
        "account suspended"
    ],
    "Prize or reward claim": [
        "you have won",
        "claim your prize",
        "free gift",
        "cash reward",
        "lottery winner"
    ],
    "Sensitive information request": [
        "share your otp",
        "send your password",
        "enter your otp",
        "bank pin",
        "verify your password"
    ],
    "Suspicious money request": [
        "pay processing fee",
        "send money to receive",
        "pay to claim",
        "transfer money immediately"
    ],
    "Unverified forwarding claim": [
        "forward this to everyone",
        "send to all contacts",
        "share with 20 people"
    ]
}

def analyze_message(message):
    text = message.lower()
    findings = []

    for category, phrases in WARNING_PATTERNS.items():
        matched = [phrase for phrase in phrases if phrase in text]
        if matched:
            findings.append(
                f"{category}: matched wording — "
                + ", ".join(matched)
            )

    if re.search(r"\b(otp|password|bank pin)\b", text):
        findings.append(
            "Sensitive credentials are mentioned. "
            "Never share OTPs, passwords or banking PINs."
        )

    if not findings:
        findings.append(
            "No configured warning patterns were detected. "
            "This does not prove the message is genuine."
        )

    if any(
        word in text
        for word in ["share your otp", "send your password",
                     "bank pin", "pay processing fee"]
    ):
        verdict = "Potential Scam"
    elif any(
        phrase in text
        for phrase in ["you have won", "claim your prize",
                       "account will be blocked", "act immediately"]
    ):
        verdict = "Suspicious — Verify"
    else:
        verdict = "Unverified"

    return verdict, findings

def analyze_link(link):
    raw_link = link.strip()
    candidate = raw_link

    if not re.match(r"^https?://", candidate, re.I):
        candidate = "https://" + candidate

    try:
        parsed = urlparse(candidate)
        host = parsed.hostname
    except ValueError:
        host = None
        parsed = None

    findings = []

    if not host or "." not in host:
        return "Invalid or Unverified", [
            "The input does not look like a valid web address."
        ]

    if parsed.scheme != "https":
        findings.append(
            "The link does not use HTTPS. "
            "This alone does not prove it is malicious."
        )

    if "@" in parsed.netloc:
        findings.append(
            "The address contains an @ symbol, which can be "
            "used to make URLs misleading."
        )

    if host.startswith("xn--") or ".xn--" in host:
        findings.append(
            "The domain uses internationalized encoding. "
            "Check the spelling carefully."
        )

    if re.match(r"^\d{1,3}(\.\d{1,3}){3}$", host):
        findings.append(
            "The link uses an IP address instead of a usual domain."
        )

    shortened_domains = {
        "bit.ly", "tinyurl.com", "t.co", "is.gd", "cutt.ly"
    }

    if host.lower() in shortened_domains:
        findings.append(
            "This is a shortened link. Its final destination "
            "is not visible from the text alone."
        )

    if host.count("-") >= 3 or len(host) > 45:
        findings.append(
            "The domain has unusual formatting or length. "
            "Check it carefully."
        )

    if not findings:
        findings.append(
            "No basic URL warning patterns were detected. "
            "The link has not been proven safe."
        )

    if any("shortened link" in item.lower() for item in findings):
        verdict = "Caution — Destination Unknown"
    elif any(
        "unusual formatting" in item.lower()
        or "misleading" in item.lower()
        or "IP address" in item
        for item in findings
    ):
        verdict = "Suspicious — Verify"
    elif parsed.scheme != "https":
        verdict = "Caution — Check Link"
    else:
        verdict = "Unverified"

    return verdict, findings

# ---------- RESULTS ----------

def calculate_risk(verdict, findings):
    text = " ".join(findings).lower()

    if verdict == "Potential Scam":
        return 90
    elif verdict == "Suspicious — Verify":
        return 75
    elif verdict == "Caution — Destination Unknown":
        return 65
    elif verdict == "Caution — Check Link":
        return 55
    elif verdict == "Invalid or Unverified":
        return 50
    elif "no configured warning patterns" in text:
        return 15
    else:
        return 30
def show_risk_meter(score):
    if score >= 70:
        color = "#ff4b4b"
        label = "High estimated risk"
    elif score >= 40:
        color = "#ffb020"
        label = "Moderate estimated risk"
    else:
        color = "#21c55d"
        label = "Low detected risk"

    st.markdown(
        f"""
        <div style="text-align:center; margin:25px auto; max-width:280px;">
          <div style="
              width:180px;height:180px;border-radius:50%;
              margin:auto;display:flex;align-items:center;
              justify-content:center;flex-direction:column;
              background:conic-gradient(
                  {color} {score * 3.6}deg,
                  #30364d 0deg
              );
          ">
            <div style="
                width:145px;height:145px;border-radius:50%;
                background:#10162f;display:flex;
                align-items:center;justify-content:center;
                flex-direction:column;color:white;
            ">
              <span style="font-size:38px;font-weight:bold;">
                {score}%
              </span>
              <span style="font-size:12px;">RISK SCORE</span>
            </div>
          </div>
          <h3 style="color:{color};margin-top:15px;">{label}</h3>
          <p style="font-size:12px;color:#cccccc;">
            Preliminary estimate from configured warning rules;
            not a verified probability that a claim is false.
          </p>
        </div>
        """,
        unsafe_allow_html=True
    )

def show_result(check_type, content, verdict, findings):
    explanation = "\n".join(findings)
    score = calculate_risk(verdict, findings)
    show_risk_meter(score)
    if verdict in ["Potential Scam", "Suspicious — Verify",
                   "Caution — Check Link",
                   "Caution — Destination Unknown",
                   "Invalid or Unverified"]:
        st.warning(f"Assessment: {verdict}")
    else:
        st.info(f"Assessment: {verdict}")

    st.markdown("### Why?")
    for finding in findings:
        st.write("•", finding)

    st.caption(
        "This is a preliminary heuristic assessment, not a "
        "confirmed fact-check or a guarantee of link safety."
    )

    save_check(check_type, content, verdict, explanation)
    st.success("Assessment saved to local history.")
